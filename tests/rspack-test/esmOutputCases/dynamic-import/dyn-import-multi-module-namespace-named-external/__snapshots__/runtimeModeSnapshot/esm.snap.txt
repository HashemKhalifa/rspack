it("should avoid namespace and named binding collisions for duplicate externals", async () => {
	const modA = await import("./shared.mjs").then(m => m.a_namespaceObject);
	const modB = await import("./shared.mjs").then(m => m.b_namespaceObject);

	expect(modA.resolve).toBeDefined();
	expect(typeof modA.resolve.join).toBe("function");
	expect(modA.resolve.resolve).toBe(modB.resolve);
	expect(modB.marker).toBe("b");
});

var __rspack_define_property_getters = (exports, getters, values) => {
	var define = (defs, kind) => {
		for(var key in defs) {
			if(__rspack_has_own_property(defs, key) && !__rspack_has_own_property(exports, key)) {
				Object.defineProperty(exports, key, { enumerable: true, [kind]: defs[key] });
			}
		}
	};

	define(values, "value");
};;

	if(typeof Symbol !== 'undefined' && Symbol.toStringTag) {
