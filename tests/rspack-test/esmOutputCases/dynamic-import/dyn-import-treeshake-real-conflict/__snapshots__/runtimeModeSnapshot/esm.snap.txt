```mjs title=main.mjs
// ./index.js
it('should keep namespace when both conflicting exports are used', async () => {
	// Both modules export `value` and both are used — real conflict.
	// Namespace objects are required to disambiguate.
	const { value: aValue, foo } = await import("./shared.mjs").then(m => m.a_namespaceObject)
	const { value: bValue, bar } = await import("./shared.mjs").then(m => m.b_namespaceObject)

	expect(aValue).toBe(1)
	expect(bValue).toBe(2)
	expect(foo).toBe('a-foo')
	expect(bar).toBe('b-bar')
})

export {};
```

```mjs title=runtime.mjs
// The require scope
var __rspack_require = {};
export { __rspack_require };

// rspack/runtime/define_property_getters
var __rspack_define_property_getters = (exports, getters, values) => {
	var define = (defs, kind) => {
		for(var key in defs) {
			if(__rspack_has_own_property(defs, key) && !__rspack_has_own_property(exports, key)) {
				Object.defineProperty(exports, key, { enumerable: true, [kind]: defs[key] });
			}
		}
	};
	define(getters, "get");
	define(values, "value");
};;
export { __rspack_define_property_getters };
// rspack/runtime/has_own_property
var __rspack_has_own_property = (obj, prop) => (Object.prototype.hasOwnProperty.call(obj, prop));
// rspack/runtime/make_namespace_object
// define __esModule on exports
var __rspack_make_namespace_object = (exports) => {
	if(typeof Symbol !== 'undefined' && Symbol.toStringTag) {
		Object.defineProperty(exports, Symbol.toStringTag, { value: 'Module' });
	}
	Object.defineProperty(exports, '__esModule', { value: true });
};;
export { __rspack_make_namespace_object };
```

```mjs title=shared.mjs
import { __rspack_require, __rspack_define_property_getters, __rspack_make_namespace_object } from "./runtime.mjs";
// NAMESPACE OBJECT: ./a.js
var a_namespaceObject = {};
__rspack_make_namespace_object(a_namespaceObject);
__rspack_define_property_getters(a_namespaceObject, {
  foo: () => (foo),
  value: () => (value) });


// NAMESPACE OBJECT: ./b.js
var b_namespaceObject = {};
__rspack_make_namespace_object(b_namespaceObject);
__rspack_define_property_getters(b_namespaceObject, {
  bar: () => (bar),
  value: () => (b_value) });


// ./a.js
const value = 1;
const foo = 'a-foo';

// ./b.js
const b_value = 2;
const bar = 'b-bar';

export { a_namespaceObject, b_namespaceObject };
```
