var __rspack_modules = {};
// The module cache
var __rspack_module_cache = {};
// The require function
function __rspack_require(moduleId) {
// Check if module is in cache
var cachedModule = __rspack_module_cache[moduleId];
if (cachedModule !== undefined) {
return cachedModule.exports;
}
// Create a new module (and put it into the cache)
var module = (__rspack_module_cache[moduleId] = {
exports: {}
});
// Execute the module function
__rspack_modules[moduleId](module, module.exports, __rspack_require);

// Return the exports of the module
return module.exports;
}
export { __rspack_require };
// expose the modules object (__rspack_modules)
var __rspack_module_factories = __rspack_modules;
export { __rspack_module_factories };

// rspack/runtime/define_property_getters
var __rspack_define_property_getters = (exports, getters, values) => {
	var define = (defs, kind) => {
		for(var key in defs) {
			if(__rspack_has_own_property(defs, key) && !__rspack_has_own_property(exports, key)) {
				Object.defineProperty(exports, key, { enumerable: true, [kind]: defs[key] });
			}
		}
	};
	define(getters, "get");
	define(values, "value");
};;
export { __rspack_define_property_getters };
// rspack/runtime/esm_ensure_chunk
var __rspack_ensure_chunk_handlers = {};
var __rspack_ensure_chunk = function(chunkId, fetchPriority) {
	return Promise.all(Object.keys(__rspack_ensure_chunk_handlers).reduce(function(promises, key) {
		__rspack_ensure_chunk_handlers[key](chunkId, promises, fetchPriority);
		return promises;
	}, []));
};
;
export { __rspack_ensure_chunk };
// rspack/runtime/esm_register_module
__rspack_module_factories.add = function registerModules(modules) { Object.assign(__rspack_module_factories, modules) }
;
// rspack/runtime/get javascript chunk filename
// This function allow to reference chunks
var __rspack_get_chunk_script_filename = (chunkId) => {
  // return url for filenames not based on template

  // return url for filenames based on template
  return "" + chunkId + ".mjs"
};
// rspack/runtime/has_own_property
var __rspack_has_own_property = (obj, prop) => (Object.prototype.hasOwnProperty.call(obj, prop));
// rspack/runtime/make_namespace_object
// define __esModule on exports
var __rspack_make_namespace_object = (exports) => {
	if(typeof Symbol !== 'undefined' && Symbol.toStringTag) {
		Object.defineProperty(exports, Symbol.toStringTag, { value: 'Module' });
	}
	Object.defineProperty(exports, '__esModule', { value: true });
};;
export { __rspack_make_namespace_object };
// rspack/runtime/esm_chunk_loading
var esmInstalledChunks = {};
var esmChunkMap = {
"a": function() { return import("./a.mjs"); }
};
__rspack_ensure_chunk_handlers.j = function(chunkId, promises) {
	var installedChunkData = esmInstalledChunks[chunkId];
	if(installedChunkData === 0) return;
	if(installedChunkData) {
		promises.push(installedChunkData);
		return;
	}
	var loadChunk = esmChunkMap[chunkId];
	if(!loadChunk) return;
	var promise = loadChunk().then(function() {
		esmInstalledChunks[chunkId] = 0;
	}, function(error) {
		delete esmInstalledChunks[chunkId];
		throw error;
	});
	esmInstalledChunks[chunkId] = promise;
	promises.push(promise);
};
;
