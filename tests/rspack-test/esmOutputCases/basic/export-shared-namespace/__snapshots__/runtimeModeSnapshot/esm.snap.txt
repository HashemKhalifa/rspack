// rspack/runtime/define_property_getters
var __rspack_define_property_getters = (exports, getters, values) => {
	var define = (defs, kind) => {
		for(var key in defs) {
			if(__rspack_has_own_property(defs, key) && !__rspack_has_own_property(exports, key)) {
				Object.defineProperty(exports, key, { enumerable: true, [kind]: defs[key] });
			}
		}
	};
	define(getters, "get");
	define(values, "value");
};;
export { __rspack_define_property_getters };
// rspack/runtime/has_own_property
var __rspack_has_own_property = (obj, prop) => (Object.prototype.hasOwnProperty.call(obj, prop));
// rspack/runtime/make_namespace_object
// define __esModule on exports
var __rspack_make_namespace_object = (exports) => {
	if(typeof Symbol !== 'undefined' && Symbol.toStringTag) {
		Object.defineProperty(exports, Symbol.toStringTag, { value: 'Module' });
	}
	Object.defineProperty(exports, '__esModule', { value: true });
};;
export { __rspack_make_namespace_object };
